import os
import json
import yfinance as yf
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from google import genai
from google.genai import types

app = FastAPI(title="NASDAQ AI Terminal Backend")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

client = genai.Client(api_key=os.environ.get("GEMINI_API_KEY"))

def collect_ticker_metrics(ticker: str) -> dict:
    stock = yf.Ticker(ticker)
    hist = stock.history(period="1y")
    close = hist["Close"]
    
    # 이동평균선 & 14일 RSI 계산
    ma20 = float(close.rolling(20).mean().iloc[-1])
    ma60 = float(close.rolling(60).mean().iloc[-1])
    ma120 = float(close.rolling(120).mean().iloc[-1])
    price = float(close.iloc[-1])
    prev_close = float(close.iloc[-2])
    change_pct = ((price - prev_close) / prev_close) * 100

    delta = close.diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    rsi = float((100 - (100 / (1 + (gain / loss)))).iloc[-1])

    info = stock.info
    news = [n.get("title", "") for n in (stock.news or [])[:5]]

    return {
        "ticker": ticker.upper(),
        "name": info.get("shortName", ticker.upper()),
        "price": round(price, 2),
        "changePct": round(change_pct, 2),
        "ma20": round(ma20, 2),
        "ma60": round(ma60, 2),
        "ma120": round(ma120, 2),
        "rsi": round(rsi, 1),
        "forwardPE": round(info.get("forwardPE", 0) or 0, 1),
        "revGrowth": round((info.get("revenueGrowth", 0) or 0) * 100, 1),
        "opMargin": round((info.get("operatingMargins", 0) or 0) * 100, 1),
        "news": news
    }

@app.get("/api/analyze")
def analyze_market(tickers: str = "QQQ,NVDA,AVGO,AMD,TSM,MSFT,GOOGL,AMZN,META,AAPL,QCOM,GEV,VRT,ETN,TSLA,PLTR,CRM"):
    ticker_list = [t.strip().upper() for t in tickers.split(",") if t.strip()]
    raw_data = [collect_ticker_metrics(t) for t in ticker_list]

    prompt = f"""
    당신은 월스트리트 시니어 퀀트 애널리스트입니다.
    다음 미국 증시 종목들의 실시간 지표와 뉴스를 분석하여 단기(1주~1개월), 중기(3~12개월), 장기(1~3년+) 점수(0~100)와
    투자 판단 근거(한국어)를 JSON으로 반환하세요.
    데이터: {json.dumps(raw_data, ensure_ascii=False)}
    """

    response = client.models.generate_content(
        model="gemini-2.5-flash",
        contents=prompt,
        config=types.GenerateContentConfig(response_mime_type="application/json")
    )
    return {"metrics": raw_data, "ai_analysis": json.loads(response.text)}
